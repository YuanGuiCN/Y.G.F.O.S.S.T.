// ================================================================ // pinyin-pro 加载器（优先本地，回退 CDN） // ================================================================ (function loadPinyinPro() { const LOCAL_FILE = 'pinyin-pro.min.js'; const CDN_URL = 'https://cdn.jsdelivr.net/npm/pinyin-pro@3.18.2/dist/index.js'; // 检查本地是否已有文件 function tryLoadLocal() { return new Promise((resolve, reject) => { const script = document.createElement('script'); script.src = LOCAL_FILE; script.onload = () => { console.log('✅ 使用本地 pinyin-pro'); resolve(); }; script.onerror = () => { console.log('⚠️ 本地未找到 pinyin-pro，尝试 CDN'); reject(); }; document.head.appendChild(script); }); } // 从 CDN 加载 function loadFromCDN() { return new Promise((resolve, reject) => { const script = document.createElement('script'); script.src = CDN_URL; script.onload = () => { console.log('✅ 已从 CDN 加载 pinyin-pro'); // 下载到本地供下次使用 downloadToLocal(CDN_URL); resolve(); }; script.onerror = reject; document.head.appendChild(script); }); } // 下载文件到本地 function downloadToLocal(url) { fetch(url) .then(res => res.blob()) .then(blob => { const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = LOCAL_FILE; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(a.href); console.log('📥 pinyin-pro 已下载到本地'); // 显示下载提示 if (typeof showToast === 'function') { showToast('📥 拼音库已下载，下次打开可离线使用'); } }) .catch(err => console.warn('下载失败:', err)); } // 执行加载流程 tryLoadLocal() .catch(() => loadFromCDN()) .catch(err => { console.error('❌ pinyin-pro 加载失败:', err); if (typeof showToast === 'function') { showToast('⚠️ 拼音库加载失败，拼音匹配不可用'); } }); })();
